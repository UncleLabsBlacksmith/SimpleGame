ส่งข้อความแรกหา AI

# pip install anthropic python-dotenv
from dotenv import load_dotenv
import anthropic

load_dotenv()   # อ่าน ANTHROPIC_API_KEY จาก .env
client = anthropic.Anthropic()

resp = client.messages.create(
    model="claude-sonnet-5-5",
    max_tokens=1024,
    messages=[
        {"role": "user",
         "content": "สวัสดี แนะนำตัวหน่อย"},
    ],
)
print(resp.content[0].text)
modelรุ่นของโมเดลที่จะใช้
max_tokensเพดานความยาวคำตอบ กันคำตอบยาวเกินและคุมค่าใช้จ่าย
messagesรายการข้อความในบทสนทนา แต่ละข้อความมี role และ content
resp.contentคำตอบจะมาเป็นลิสต์ของ block ข้อความอยู่ใน block แรก

ทำให้ AI จำบทสนทนาได้

history = []
SYSTEM = "คุณคือผู้ช่วยของเกม SimpleGame ตอบสั้น กระชับ"

def chat(user_text):
    history.append({"role": "user", "content": user_text})
    resp = client.messages.create(
        model="claude-sonnet-5-5", max_tokens=1024,
        system=SYSTEM,
        messages=history,     # ส่งประวัติทั้งหมดทุกครั้ง
    )
    answer = resp.content[0].text
    history.append({"role": "assistant", "content": answer})
    return answer

API ไม่จำอะไรเลย

แต่ละคำขอเป็นอิสระต่อกัน ความจำของแชทคือลิสต์ history ที่เราเก็บและส่งไปเองทุกรอบ

systemกำหนดบทบาท น้ำเสียง และกติกาของบอท
roleสลับกันระหว่าง user และ assistant

Streaming: แสดงคำตอบทีละคำ

with client.messages.stream(
    model="claude-sonnet-5-5",
    max_tokens=1024,
    system=SYSTEM,
    messages=history,
) as stream:
    for text in stream.text_stream:
        print(text, end="", flush=True)

    final = stream.get_final_message()
    print("\nใช้ไป", final.usage.output_tokens, "tokens")
  • ผู้ใช้เห็นคำแรกภายในเสี้ยววินาที แทนที่จะรอคำตอบทั้งก้อน
  • คำตอบยาว ๆ ไม่ดูเหมือนแอปค้าง
  • usage บอกจำนวน token ที่ใช้ เอาไว้บันทึกค่าใช้จ่าย

ต่อ Backend กับหน้าแชท

# app.py  →  uvicorn app:app --reload
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
from pydantic import BaseModel

app = FastAPI()

class ChatIn(BaseModel):
    messages: list[dict]

@app.post("/chat")
def chat(body: ChatIn):
    def gen():
        with client.messages.stream(
            model="claude-sonnet-5-5", max_tokens=1024,
            system=SYSTEM, messages=body.messages,
        ) as s:
            yield from s.text_stream
    return StreamingResponse(gen(), media_type="text/plain")
// chat.js (ฝั่งเบราว์เซอร์ ไม่มีคีย์)
const res = await fetch("/chat", {
  method: "POST",
  headers: { "Content-Type": "application/json" },

while (true) {

messages.push({ role: "assistant", content: answer });

คุมค่าใช้จ่ายและรับมือข้อจำกัด

คิดเงินเป็น token

นับทั้งขาเข้าและขาออก ประวัติแชทที่ส่งซ้ำทุกรอบจึงเป็นต้นทุนหลัก

ตัดหรือสรุปประวัติ

เก็บแค่ N ข้อความล่าสุด หรือให้ AI สรุปบทสนทนาเก่าเป็นย่อหน้าเดียว

เลือกรุ่นให้เหมาะงาน

รุ่นเล็ก (Haiku) สำหรับงานง่าย รุ่นกลาง (Sonnet) งานทั่วไป รุ่นใหญ่ (Opus) งานยาก

Prompt caching

system prompt หรือเอกสารยาวที่ส่งซ้ำ ๆ แคชไว้ได้ ช่วยลดทั้งเงินและเวลา

ตั้งวงเงินใน console

กำหนด spending limit และดูกราฟการใช้งานรายวัน กันบิลพุ่งจากบั๊ก

รับมือ error 429

เมื่อชน rate limit ให้รอแล้วลองใหม่ (retry แบบ backoff) และแจ้งผู้ใช้
