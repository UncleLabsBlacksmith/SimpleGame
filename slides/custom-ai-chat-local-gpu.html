resp = requests.post("http://localhost:11434/api/chat", json={
    "model": "llama3.1:8b",
    "messages": [

        {"role": "user", "content": "สวัสดี แนะนำตัวหน่อย"},

Backend แบบ streaming สำหรับหน้าแชท

import json, requests
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
from pydantic import BaseModel

app = FastAPI()
OLLAMA = "http://localhost:11434/api/chat"

class ChatIn(BaseModel):
    messages: list[dict]

@app.post("/chat")
def chat(body: ChatIn):
    def gen():
        payload = {"model": "llama3.1:8b",
                   "messages": body.messages,
                   "stream": True}
        with requests.post(OLLAMA, json=payload, stream=True) as r:
            for line in r.iter_lines():     # 1 บรรทัด = 1 JSON
                if line:
                    yield json.loads(line)["message"]["content"]
    return StreamingResponse(gen(), media_type="text/plain")
